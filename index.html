Pengaturan Auto-Save Spreadsheet

Hubungkan CBT dengan Google Sheets Anda

Status: Web App URL Tersimpan & Siap Digunakan!
Kode Google Apps Script (Copy & Paste) 
const ACCESS_KEY = 'CBT_OPEN_SUBJECTS';
const SPREADSHEET_ID = '1PQwqWeL1P3baRSsdaHRqlJnafvf2to67PEMoXNUz4Ao';
const ALLOWED_SUBJECTS = ['Matematika'];

function doGet(e) {
  const callback = ((e && e.parameter && e.parameter.callback) || '')
    .replace(/[^a-zA-Z0-9_.$]/g, '');

  const saved = PropertiesService
    .getScriptProperties()
    .getProperty(ACCESS_KEY);

  let subjects = ALLOWED_SUBJECTS;

  if (saved) {
    try {
      const parsed = JSON.parse(saved);
      subjects = Array.isArray(parsed)
        ? parsed.filter(subject => ALLOWED_SUBJECTS.includes(subject))
        : [];
    } catch (error) {
      subjects = [];
    }
  }

  const result = JSON.stringify({
    subjects: subjects,
    configured: Boolean(saved)
  });

  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + result + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  return ContentService
    .createTextOutput(result)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const data = JSON.parse(
      e && e.postData && e.postData.contents
        ? e.postData.contents
        : '{}'
    );

    if (data.action === 'saveAccess') {
      const subjects = Array.isArray(data.subjects)
        ? data.subjects.filter(subject => ALLOWED_SUBJECTS.includes(subject))
        : [];

      PropertiesService
        .getScriptProperties()
        .setProperty(ACCESS_KEY, JSON.stringify(subjects));

      return ContentService.createTextOutput('Access saved');
    }

    const sheet = SpreadsheetApp
      .openById(SPREADSHEET_ID)
      .getSheets()[0];

    sheet.appendRow([
      data.waktuSelesai || '',
      data.nama || '',
      data.kelas || '',
      data.noAbsen || '',
      data.tanggalLahir || '',
      data.mapel || '',
      data.nilai ?? '',
      data.ringkasan || ''
    ]);

    return ContentService.createTextOutput('Success');
  } catch (error) {
    return ContentService.createTextOutput('Error: ' + error.message);
  }
}

*Deploy kode sebagai Web App dengan Execute as: Me dan Who has access: Anyone. URL Web App yang saat ini digunakan sudah diatur otomatis. Jika nanti membuat deployment baru, ganti URL melalui tombol pengaturan.
